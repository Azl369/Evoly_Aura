cmake_minimum_required(VERSION 3.24)
project(EvolyAura VERSION 0.0.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "P0 currently requires Windows and MSVC.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /permissive- /utf-8 /EHsc)
add_library(aura_core STATIC src/core/image.cpp src/core/workspace.cpp)
target_include_directories(aura_core PUBLIC include)
add_library(aura_render STATIC src/render/renderer.cpp src/render/workspace_renderer.cpp src/platform/windows/wallpaper.cpp)
target_include_directories(aura_render PUBLIC include)
target_link_libraries(aura_render PUBLIC aura_core d2d1 dwrite d3d11 dxgi dcomp windowscodecs ole32 shell32 dwmapi)
add_executable(AuraProbe WIN32 apps/host/main.cpp apps/host/app.manifest src/platform/windows/probe.cpp)
target_link_libraries(AuraProbe PRIVATE aura_render comctl32)
add_executable(Aura WIN32 apps/host/aura_main.cpp apps/host/app.manifest src/platform/windows/desktop_app.cpp src/platform/windows/system_sampler.cpp src/persistence/workspace_store.cpp)
target_link_libraries(Aura PRIVATE aura_render comctl32 pdh)
add_executable(AuraMetrics tools/native/metrics.cpp)
target_link_libraries(AuraMetrics PRIVATE psapi pdh)
include(CTest)
if(BUILD_TESTING)
  add_executable(AuraCoreTests tests/unit/image_test.cpp)
  target_link_libraries(AuraCoreTests PRIVATE aura_core)
  add_test(NAME aura_core COMMAND AuraCoreTests)
  add_executable(AuraWorkspaceTests tests/unit/workspace_test.cpp src/persistence/workspace_store.cpp)
  target_link_libraries(AuraWorkspaceTests PRIVATE aura_core)
  add_test(NAME aura_workspace COMMAND AuraWorkspaceTests "${CMAKE_BINARY_DIR}/test-state")
  add_executable(AuraSystemTests tests/unit/system_sampler_test.cpp src/platform/windows/system_sampler.cpp apps/host/app.manifest)
  target_link_libraries(AuraSystemTests PRIVATE aura_render pdh)
  add_test(NAME aura_system_sampler COMMAND AuraSystemTests)
endif()
